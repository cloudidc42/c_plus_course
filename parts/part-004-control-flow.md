## 4.1 if, else if, else และแนวคิดเรื่อง "จริง/เท็จ" ใน C (Step 25)

โครงสร้าง `if` คือคำสั่งควบคุมการทำงาน (Control Flow) ที่พื้นฐานที่สุด ให้โปรแกรมเลือก
ทำงานตามเงื่อนไขที่กำหนด

```c
/* if_basic.c */
#include <stdio.h>

int main(void) {
    int temperature = 35;

    if (temperature > 30) {
        printf("อากาศร้อนมาก ควรดื่มน้ำเยอะๆ\n");
    }

    printf("โปรแกรมทำงานต่อ ไม่ว่าเงื่อนไขจะจริงหรือเท็จ\n");

    return 0;
}
```

### C ไม่มี "boolean" แท้ๆ (ก่อน C99) — ทุกอย่างคือตัวเลข

หัวใจสำคัญที่ต้องเข้าใจคือ C ตัดสิน "จริง/เท็จ" จากค่าตัวเลขโดยตรง **ไม่ใช่ศูนย์ = จริง,
ศูนย์ = เท็จ** เสมอ ไม่ว่าเงื่อนไขนั้นจะเป็นนิพจน์เปรียบเทียบ (`x > 5`) หรือค่าตัวแปรตรงๆ
ก็ตาม

```c
/* truthy_falsy.c */
#include <stdio.h>

int main(void) {
    int values[] = {0, 1, -1, 42, 0};

    for (size_t i = 0; i < sizeof(values) / sizeof(values[0]); i++) {
        if (values[i]) {   /* เทียบเท่ากับ if (values[i] != 0) */
            printf("%d ถือว่าเป็น \"จริง\"\n", values[i]);
        } else {
            printf("%d ถือว่าเป็น \"เท็จ\"\n", values[i]);
        }
    }

    return 0;
}
```

ผลลัพธ์:

```
0 ถือว่าเป็น "เท็จ"
1 ถือว่าเป็น "จริง"
-1 ถือว่าเป็น "จริง"
42 ถือว่าเป็น "จริง"
0 ถือว่าเป็น "เท็จ"
```

สังเกตว่า `-1` ก็ถือว่า "จริง" เช่นกัน เพราะกฎของ C คือ **ค่าใดๆ ที่ไม่ใช่ 0 ถือว่าจริงหมด**
ไม่ใช่แค่ค่าบวก — นี่เป็นเหตุผลที่ฟังก์ชันมาตรฐานหลายตัวใช้ `-1` เป็นค่าแสดง error (เพื่อให้
`if (function_call())` ยังคงเข้าเงื่อนไข "จริง" ได้ถ้าต้องการตรวจแค่ "สำเร็จหรือไม่สำเร็จ"
แต่ในทางปฏิบัติควรตรวจสอบให้ชัดเจนเสมอว่าค่าที่คืนกลับมาหมายถึงอะไรกันแน่)

### ตั้งแต่ C99: `<stdbool.h>` ทำให้เจตนาชัดเจนขึ้น

```c
/* stdbool_demo.c */
#include <stdio.h>
#include <stdbool.h>

int main(void) {
    int  age = 20;
    bool is_adult = (age >= 18);

    if (is_adult) {
        printf("เป็นผู้ใหญ่แล้ว\n");
    } else {
        printf("ยังไม่บรรลุนิติภาวะ\n");
    }

    printf("is_adult มีค่าจริงๆ คือ %d (bool เป็นแค่ alias ของ _Bool ซึ่งเก็บ 0/1)\n",
           is_adult);

    return 0;
}
```

`bool`, `true`, `false` ใน `<stdbool.h>` เป็นเพียง Macro ที่ขยายเป็น `_Bool`, `1`, `0`
ตามลำดับ (ชนิด `_Bool` ถูกเพิ่มเข้ามาจริงๆ ในมาตรฐาน C99) แต่การใช้ `bool` แทน `int` เมื่อ
ตัวแปรมีความหมายเป็นจริง/เท็จอย่างชัดเจน ทำให้โค้ดสื่อเจตนาได้ดีขึ้นมากและเป็นธรรมเนียม
ที่แนะนำอย่างยิ่งตั้งแต่ C99 เป็นต้นมา

### if-else if-else แบบครบ

```c
/* if_else_chain.c */
#include <stdio.h>

int main(void) {
    int score = 78;

    if (score >= 90) {
        printf("เกรด A\n");
    } else if (score >= 80) {
        printf("เกรด B\n");
    } else if (score >= 70) {
        printf("เกรด C\n");
    } else if (score >= 60) {
        printf("เกรด D\n");
    } else {
        printf("เกรด F\n");
    }

    return 0;
}
```

ข้อสังเกตสำคัญ: **ลำดับของเงื่อนไขมีผลต่อความถูกต้อง** เพราะ `else if` แต่ละอันจะถูก
ตรวจสอบตามลำดับจากบนลงล่าง และหยุดทันทีที่เจอเงื่อนไขแรกที่เป็นจริง ถ้าเขียนสลับลำดับ
เป็นจากน้อยไปมาก (`score >= 60` ก่อน) เกรดของทุกคนที่ควรได้ B หรือ A จะกลายเป็น D หมด
เพราะเงื่อนไขแรกที่เจอ (`>= 60`) จะจริงและหยุดตรวจก่อนถึงเงื่อนไขที่ถูกต้อง

## 4.2 ออกแบบเงื่อนไขให้อ่านง่าย: Guard Clause (Step 26)

เมื่อเงื่อนไขซ้อนกันหลายชั้น (Nested If) โค้ดจะอ่านยากขึ้นเรื่อยๆ ตามความลึกของการซ้อน
เทคนิค **Guard Clause** คือการตรวจสอบเงื่อนไข "ที่ทำให้ต้องออกจากฟังก์ชันทันที" ไว้ก่อน
ที่บนสุดของฟังก์ชัน แทนที่จะซ้อน `if` ลึกลงไปเรื่อยๆ

```c
/* nested_if_bad.c — ตัวอย่างโค้ดที่ซ้อนลึกเกินไป (ไม่แนะนำ) */
#include <stdio.h>
#include <stdbool.h>

bool can_watch_movie_nested(int age, bool has_ticket, bool is_theater_open) {
    if (is_theater_open) {
        if (has_ticket) {
            if (age >= 18) {
                return true;
            } else {
                printf("อายุไม่ถึงเกณฑ์\n");
                return false;
            }
        } else {
            printf("ไม่มีตั๋ว\n");
            return false;
        }
    } else {
        printf("โรงหนังปิด\n");
        return false;
    }
}

int main(void) {
    printf("%s\n", can_watch_movie_nested(20, true, true) ? "เข้าชมได้" : "เข้าชมไม่ได้");
    return 0;
}
```

โค้ดข้างบนทำงานถูกต้อง แต่การอ่านต้องไล่ตามวงเล็บปีกกาซ้อนกันถึง 3 ชั้น ทำให้ตามตรรกะยาก
ขึ้นเรื่อยๆ เมื่อเงื่อนไขมีมากกว่านี้ ลองเขียนใหม่ด้วย Guard Clause:

```c
/* guard_clause_good.c — ใช้ Guard Clause ทำให้อ่านง่ายขึ้นมาก */
#include <stdio.h>
#include <stdbool.h>

bool can_watch_movie(int age, bool has_ticket, bool is_theater_open) {
    if (!is_theater_open) {
        printf("โรงหนังปิด\n");
        return false;
    }
    if (!has_ticket) {
        printf("ไม่มีตั๋ว\n");
        return false;
    }
    if (age < 18) {
        printf("อายุไม่ถึงเกณฑ์\n");
        return false;
    }

    return true;   /* ผ่านทุกเงื่อนไขแล้ว จึงมาถึงจุดนี้ */
}

int main(void) {
    printf("%s\n", can_watch_movie(20, true, true) ? "เข้าชมได้" : "เข้าชมไม่ได้");
    return 0;
}
```

ทั้งสองฟังก์ชันทำงานเหมือนกันทุกประการ แต่เวอร์ชัน Guard Clause อ่านง่ายกว่ามาก เพราะ
**แต่ละ `if` ตรวจสอบแค่เงื่อนไขเดียวและจบด้วย `return` ทันที** ไม่มีการซ้อน `else` ลึกลง
ไปเรื่อยๆ ผู้อ่านสามารถอ่านจากบนลงล่างและเข้าใจ "กฎการตัดสิทธิ์" ทีละข้อได้ทันที โดยไม่
ต้องจำสถานะของ `if` ชั้นบนไว้ในหัวขณะไล่อ่านโค้ดชั้นล่าง — เทคนิคนี้เป็นหนึ่งใน Coding
Practice ที่ใช้กันแพร่หลายที่สุดในบริษัทซอฟต์แวร์มืออาชีพ

### หลักการออกแบบเงื่อนไขให้อ่านง่าย

1. **จำกัดความลึกของการซ้อน `if` ไม่เกิน 2-3 ชั้น** — ถ้าลึกกว่านี้ ให้พิจารณาแยกเป็น
   ฟังก์ชันย่อย หรือใช้ Guard Clause
2. **ตั้งชื่อตัวแปร boolean ให้สื่อความหมายเป็นประโยคคำถาม** เช่น `is_valid`, `has_error`,
   `can_proceed` แทนชื่อกำกวมอย่าง `flag`, `check`
3. **หลีกเลี่ยงการปฏิเสธซ้อนปฏิเสธ** เช่น `if (!is_not_ready)` อ่านยากกว่า `if (is_ready)`
   มาก ควรตั้งชื่อตัวแปรในรูปแบบบวก (Positive Form) เสมอเมื่อทำได้
4. **จัดกลุ่มเงื่อนไขที่เกี่ยวข้องกันด้วยวงเล็บและช่องว่างให้ชัดเจน** โดยเฉพาะเมื่อผสม
   `&&` กับ `||` ในนิพจน์เดียวกัน (จะเจาะลึกในหัวข้อ 4.6)

## 4.3 switch-case พื้นฐาน: Fallthrough และ break (Step 27)

`switch` เหมาะกับกรณีที่ต้องเปรียบเทียบตัวแปรตัวเดียวกับค่าคงที่หลายค่า แทนการเขียน
`if-else if` ยาวๆ ที่เปรียบเทียบตัวแปรเดิมซ้ำไปซ้ำมา

```c
/* switch_basic.c */
#include <stdio.h>

int main(void) {
    int day = 3;

    switch (day) {
        case 1:
            printf("วันจันทร์\n");
            break;
        case 2:
            printf("วันอังคาร\n");
            break;
        case 3:
            printf("วันพุธ\n");
            break;
        case 4:
            printf("วันพฤหัสบดี\n");
            break;
        case 5:
            printf("วันศุกร์\n");
            break;
        case 6:
        case 7:
            printf("วันหยุดสุดสัปดาห์\n");
            break;
        default:
            printf("ค่าวันไม่ถูกต้อง (ต้องเป็น 1-7)\n");
            break;
    }

    return 0;
}
```

`switch` ทำงานกับชนิดข้อมูลจำนวนเต็มเท่านั้น (`int`, `char`, `enum` — จะเรียนเรื่อง `enum`
ใน Part 10) **ใช้กับ `float`, `double`, หรือ `string` ไม่ได้** เพราะมาตรฐานภาษากำหนดไว้
ให้ `switch` เปรียบเทียบด้วยค่าคงที่จำนวนเต็มที่รู้ผลตั้งแต่ตอน compile เท่านั้น

### Fallthrough: พฤติกรรมที่ทั้งมีประโยชน์และอันตราย

**Fallthrough** คือพฤติกรรมที่ถ้า `case` ใด**ไม่มี `break`** การทำงานจะ "ไหลต่อ" ไปยัง
`case` ถัดไปโดยอัตโนมัติ **โดยไม่ตรวจสอบเงื่อนไขของ `case` ถัดไปอีก**

```c
/* fallthrough_bug.c — ตัวอย่างบั๊กจากการลืม break */
#include <stdio.h>

int main(void) {
    int day = 3;

    switch (day) {
        case 1:
            printf("วันจันทร์\n");
        case 2:                      /* ลืม break จากบรรทัดบน! */
            printf("วันอังคาร\n");
        case 3:                      /* ลืม break อีก! */
            printf("วันพุธ\n");
            break;
        default:
            printf("วันอื่น\n");
    }

    return 0;
}
```

```bash
gcc -Wall -Wextra -Wpedantic -std=c17 fallthrough_bug.c -o fallthrough_bug
```

`-Wextra` มี flag `-Wimplicit-fallthrough` ซ้อนอยู่ ซึ่งจะจับกรณีนี้ได้ทันที:

```
warning: this statement may fall through [-Wimplicit-fallthrough=]
    6 |             printf("วันจันทร์\n");
      |             ^~~~~~~~~~~~~~~~~~
    7 |         case 2:
      |         ^~~~
```

และถ้า `day` เป็น `1` โปรแกรมจะพิมพ์ทั้ง "วันจันทร์", "วันอังคาร" และ "วันพุธ" ต่อกันหมด
(ไหลผ่าน case 2 และ 3 เพราะไม่มี `break` คั่น) ทั้งที่ผู้เขียนน่าจะตั้งใจให้พิมพ์แค่บรรทัด
เดียว — **การลืม `break` คือบั๊กที่พบบ่อยที่สุดของ `switch-case`** ในภาษา C และเป็นเหตุผล
สำคัญข้อหนึ่งที่กฎทองของหลักสูตรนี้ยืนยันให้เปิด `-Wall -Wextra` เสมอ เพราะ Warning นี้
จับบั๊กประเภทนี้ได้เกือบทุกกรณี แนวทางที่ปลอดภัยที่สุดคือ **ตรวจสอบให้แน่ใจว่าทุก `case`
ที่ไม่ได้ตั้งใจให้ fallthrough ต้องมี `break` เสมอ** และไม่เพิกเฉยต่อ Warning นี้เด็ดขาด

### Fallthrough แบบตั้งใจ (Intentional Fallthrough)

Fallthrough ไม่ใช่ "บั๊กเสมอไป" — บางครั้งเราต้องการให้หลาย `case` ทำงานแบบเดียวกันโดย
ตั้งใจ ดังตัวอย่าง `case 6: case 7:` ในโค้ดแรกของหัวข้อนี้ (case ที่ไม่มี statement ใดๆ
คั่นระหว่างกันจะไหลต่อกันไปโดยธรรมชาติ ซึ่งถือเป็นรูปแบบที่ชัดเจนและปลอดภัย)

```c
/* intentional_fallthrough.c */
#include <stdio.h>

int main(void) {
    char grade = 'B';

    switch (grade) {
        case 'A':
        case 'B':
        case 'C':
            printf("สอบผ่าน\n");
            break;
        case 'D':
        case 'F':
            printf("สอบไม่ผ่าน\n");
            break;
        default:
            printf("เกรดไม่ถูกต้อง\n");
            break;
    }

    return 0;
}
```

เมื่อ Fallthrough เป็นความตั้งใจแต่มี statement คั่นอยู่ (ไม่ใช่แค่ case ว่างๆ ต่อกัน)
ธรรมเนียมที่ดีคือใส่ comment กำกับชัดเจนว่า `/* fallthrough */` เพื่อบอกผู้อ่านโค้ด
(และเครื่องมือ Static Analysis) ว่านี่ไม่ใช่บั๊ก แต่เป็นการตั้งใจ:

```c
switch (value) {
    case 1:
        printf("กำลังประมวลผลระดับ 1\n");
        /* fallthrough */
    case 2:
        printf("กำลังประมวลผลระดับ 2 (รวมถึงกรณีที่มาจาก case 1 ด้วย)\n");
        break;
    default:
        break;
}
```

## 4.4 switch-case ขั้นสูง: ข้อจำกัดและการออกแบบ (Step 28)

### switch ต้องมี default เสมอ (แม้จะดูไม่จำเป็น)

```c
/* switch_default_matters.c */
#include <stdio.h>

int main(void) {
    int choice = 99;   /* ค่าที่ไม่ตรงกับ case ใดเลย */

    switch (choice) {
        case 1:
            printf("ตัวเลือกที่ 1\n");
            break;
        case 2:
            printf("ตัวเลือกที่ 2\n");
            break;
        default:
            printf("ตัวเลือกไม่ถูกต้อง กรุณาเลือกใหม่\n");
            break;
    }

    return 0;
}
```

การใส่ `default` ไว้เสมอ (แม้จะคิดว่า "ไม่มีทางเกิดค่านอกเหนือจากที่ระบุ") เป็นแนวปฏิบัติ
ที่ดี เพราะป้องกันกรณีที่ข้อมูล input ผิดพลาดจากที่คาดไว้ (เช่น จากผู้ใช้ หรือจากบั๊กใน
ส่วนอื่นของโปรแกรม) ไม่ให้ผ่านไปแบบเงียบๆ โดยไม่มีการแจ้งเตือนใดๆ

### case ต้องเป็นค่าคงที่ที่รู้ผลตอน compile เท่านั้น

```c
/* switch_constant_only.c — ตัวอย่างที่ "คอมไพล์ไม่ผ่าน" ห้ามเขียนแบบนี้ */
int main(void) {
    int lower = 1, upper = 10;
    int x = 5;

    switch (x) {
        case lower:      /* ERROR: case label ต้องเป็น constant expression เท่านั้น */
        case upper:      /* ERROR เช่นกัน */
            break;
    }
    return 0;
}
```

โค้ดข้างบน**คอมไพล์ไม่ผ่าน** เพราะ `lower` และ `upper` เป็นตัวแปร (Runtime Value) ไม่ใช่
ค่าคงที่ (Compile-time Constant) — กฎนี้คือความแตกต่างสำคัญระหว่าง `switch` กับ `if`:
`if` เปรียบเทียบกับนิพจน์อะไรก็ได้ที่ประเมินผลได้ตอนรัน แต่ `switch` ต้องเทียบกับค่าคงที่
ที่ compiler รู้ผลล่วงหน้าเท่านั้น ถ้าต้องการเปรียบเทียบกับค่าที่เป็นตัวแปรหรือช่วงค่า
(Range) ต้องใช้ `if-else if` แทน

### เมื่อไรควรใช้ switch แทน if-else if

| สถานการณ์ | แนะนำให้ใช้ |
|---|---|
| เปรียบเทียบตัวแปรตัวเดียวกับค่าคงที่จำนวนเต็ม/ตัวอักษรหลายค่า | `switch` (อ่านง่ายกว่า เร็วกว่าในบาง compiler เพราะอาจ optimize เป็น jump table) |
| เปรียบเทียบกับช่วงค่า (Range) เช่น `score >= 80` | `if-else if` (switch ทำแบบนี้ไม่ได้) |
| เงื่อนไขซับซ้อนที่รวมตัวแปรหลายตัว เช่น `age >= 18 && has_id` | `if-else if` |
| เปรียบเทียบกับค่า `float`/`double`/`string` | `if-else if` (switch ใช้ไม่ได้เลย) |

## 4.5 Ternary Operator: เงื่อนไขแบบย่อ (Step 29)

Ternary Operator (`condition ? value_if_true : value_if_false`) เป็นตัวดำเนินการเดียว
ในภาษา C ที่รับ Operand ถึง 3 ตัว (จึงเรียกว่า Ternary) ใช้แทน `if-else` แบบสั้นๆ ที่
**คืนค่าเป็นนิพจน์** ได้

```c
/* ternary_basic.c */
#include <stdio.h>

int main(void) {
    int age = 20;

    /* แบบ if-else ปกติ */
    const char *status1;
    if (age >= 18) {
        status1 = "ผู้ใหญ่";
    } else {
        status1 = "ผู้เยาว์";
    }
    printf("แบบ if-else : %s\n", status1);

    /* แบบ ternary - บรรทัดเดียวจบ */
    const char *status2 = (age >= 18) ? "ผู้ใหญ่" : "ผู้เยาว์";
    printf("แบบ ternary : %s\n", status2);

    /* ใช้ inline ใน printf ได้โดยตรง */
    printf("อายุ %d ปี ถือว่าเป็น%s\n", age, (age >= 18) ? "ผู้ใหญ่" : "ผู้เยาว์");

    return 0;
}
```

### หาค่ามากสุด/น้อยสุดด้วย Ternary

```c
/* ternary_max.c */
#include <stdio.h>

int main(void) {
    int a = 15, b = 27;

    int max = (a > b) ? a : b;
    int min = (a < b) ? a : b;

    printf("max(%d, %d) = %d\n", a, b, max);
    printf("min(%d, %d) = %d\n", a, b, min);

    return 0;
}
```

### ข้อควรระวัง: อย่าซ้อน Ternary หลายชั้นจนอ่านไม่ออก

```c
/* ternary_nested_bad.c — อ่านยากเกินไป ไม่แนะนำ */
#include <stdio.h>

int main(void) {
    int score = 78;

    /* ซ้อน ternary 3 ชั้น - อ่านยากมาก แม้จะทำงานถูกต้อง */
    char grade = (score >= 90) ? 'A' : (score >= 80) ? 'B' : (score >= 70) ? 'C' : 'F';

    printf("เกรด: %c\n", grade);

    return 0;
}
```

โค้ดข้างบนทำงานถูกต้อง แต่การซ้อน Ternary หลายชั้นแบบนี้ทำให้อ่านตรรกะได้ยากมาก โดยเฉพาะ
เมื่อเงื่อนไขมีมากกว่า 2-3 ระดับ **กฎทองของ Ternary คือ: ใช้เฉพาะกรณีเงื่อนไขเดียว ผลลัพธ์
สั้นๆ ที่อ่านจบในบรรทัดเดียวได้อย่างชัดเจน** ถ้าเงื่อนไขซับซ้อนกว่านั้น ให้กลับไปใช้
`if-else if` แบบเต็มรูปแบบเสมอ แม้จะเขียนยาวกว่าก็ตาม เพราะความอ่านง่ายสำคัญกว่าความสั้น

## 4.6 เงื่อนไขซับซ้อนและ De Morgan's Law (Step 30)

เมื่อรวมเงื่อนไขหลายตัวด้วย `&&`, `||`, `!` บางครั้งนิพจน์ที่ได้อ่านยากและเข้าใจผิดง่าย
**De Morgan's Law** เป็นกฎทางตรรกศาสตร์ที่ช่วยแปลงนิพจน์ปฏิเสธของ AND/OR ให้อยู่ในรูปที่
อ่านง่ายขึ้น:

```
!(A && B)  เทียบเท่ากับ  (!A || !B)
!(A || B)  เทียบเท่ากับ  (!A && !B)
```

```c
/* de_morgan_demo.c */
#include <stdio.h>
#include <stdbool.h>

int main(void) {
    bool has_ticket = true;
    bool is_vip     = false;

    /* รูปแบบที่ 1: เขียนแบบปฏิเสธซ้อน (อ่านยากกว่า) */
    if (!(has_ticket && is_vip)) {
        printf("แบบที่ 1: ไม่สามารถเข้าห้อง VIP ได้\n");
    }

    /* รูปแบบที่ 2: ใช้ De Morgan's Law แปลงเป็นรูปที่อ่านง่ายกว่า (ผลลัพธ์เหมือนกันทุกกรณี) */
    if (!has_ticket || !is_vip) {
        printf("แบบที่ 2: ไม่สามารถเข้าห้อง VIP ได้ (ผลลัพธ์ตรงกับแบบที่ 1 เสมอ)\n");
    }

    return 0;
}
```

### ตัวอย่างที่ใช้บ่อยในโค้ดจริง: การตรวจสอบ Input ที่ถูกต้อง

```c
/* de_morgan_practical.c */
#include <stdio.h>
#include <stdbool.h>

int main(void) {
    int score = 105;   /* ค่าที่ผิดพลาด ควรอยู่ระหว่าง 0-100 */

    /* รูปแบบปฏิเสธ: "ถ้าไม่ใช่ว่า (อยู่ในช่วง 0 ถึง 100)" */
    if (!(score >= 0 && score <= 100)) {
        printf("รูปแบบที่ 1 (ปฏิเสธ AND): คะแนนไม่ถูกต้อง\n");
    }

    /* แปลงด้วย De Morgan: !(A && B) -> (!A || !B) */
    /* !(score >= 0)  เทียบเท่ากับ  score < 0 */
    /* !(score <= 100) เทียบเท่ากับ score > 100 */
    if (score < 0 || score > 100) {
        printf("รูปแบบที่ 2 (De Morgan แล้ว): คะแนนไม่ถูกต้อง\n");
    }

    return 0;
}
```

รูปแบบที่ 2 (`score < 0 || score > 100`) อ่านเข้าใจง่ายกว่ารูปแบบที่ 1
(`!(score >= 0 && score <= 100)`) อย่างชัดเจน แม้ทั้งสองจะให้ผลลัพธ์ตรงกันทุกกรณีตาม
กฎ De Morgan ก็ตาม — ธรรมเนียมที่ดีคือ **หลีกเลี่ยงเครื่องหมาย `!` ครอบนิพจน์ที่มี `&&`
หรือ `||` อยู่ข้างในเสมอที่ทำได้** เพราะสมองมนุษย์ตีความนิพจน์เชิงบวก (Positive Logic)
ได้เร็วกว่านิพจน์ที่มีการปฏิเสธซ้อนกันมาก

## 4.7 ตัวอย่างโปรแกรมจริง: เครื่องคิดเกรด (Step 31)

มารวมทุกสิ่งที่เรียนมาสร้างโปรแกรมเครื่องคิดเกรดที่สมบูรณ์ ตรวจสอบ input ให้ถูกต้อง และ
ออกแบบเงื่อนไขด้วย Guard Clause:

```c
/* grade_calculator.c */
#include <stdio.h>
#include <stdbool.h>

/* คืนค่าเกรดตัวอักษรจากคะแนน (สมมติว่า score ผ่านการตรวจสอบแล้วว่าอยู่ในช่วง 0-100) */
char score_to_grade(double score) {
    if (score >= 90.0) {
        return 'A';
    } else if (score >= 80.0) {
        return 'B';
    } else if (score >= 70.0) {
        return 'C';
    } else if (score >= 60.0) {
        return 'D';
    } else {
        return 'F';
    }
}

/* ตรวจสอบว่าคะแนนอยู่ในช่วงที่ถูกต้องหรือไม่ */
bool is_valid_score(double score) {
    return (score >= 0.0 && score <= 100.0);
}

void clear_input_buffer(void) {
    int c;
    while ((c = getchar()) != '\n' && c != EOF) {
        /* discard */
    }
}

int main(void) {
    double score;

    printf("=== โปรแกรมคำนวณเกรด ===\n");
    printf("กรุณาใส่คะแนน (0-100): ");

    while (scanf("%lf", &score) != 1) {
        printf("ข้อมูลไม่ถูกต้อง กรุณาใส่ตัวเลขเท่านั้น: ");
        clear_input_buffer();
    }
    clear_input_buffer();

    /* Guard Clause: ตรวจสอบเงื่อนไขผิดพลาดก่อน แล้วออกทันที */
    if (!is_valid_score(score)) {
        printf("คะแนนต้องอยู่ระหว่าง 0-100 เท่านั้น (ได้รับ %.2f)\n", score);
        return 1;   /* exit code ไม่เท่ากับ 0 แสดงว่าโปรแกรมจบแบบผิดพลาด */
    }

    char grade = score_to_grade(score);
    printf("คะแนน %.2f -> เกรด %c\n", score, grade);

    /* ให้ข้อความให้กำลังใจเพิ่มเติมตามเกรด ด้วย switch */
    switch (grade) {
        case 'A':
            printf("ยอดเยี่ยมมาก!\n");
            break;
        case 'B':
        case 'C':
            printf("ทำได้ดี พยายามต่อไป!\n");
            break;
        case 'D':
            printf("ผ่านแบบเฉียดฉิว ควรทบทวนเพิ่มเติม\n");
            break;
        default:
            printf("ยังไม่ผ่านเกณฑ์ อย่าเพิ่งท้อ ลองใหม่ในเทอมหน้า!\n");
            break;
    }

    return 0;
}
```

```bash
gcc -Wall -Wextra -Wpedantic -std=c17 grade_calculator.c -o grade_calculator
echo "85.5" | ./grade_calculator
```

ผลลัพธ์:

```
=== โปรแกรมคำนวณเกรด ===
กรุณาใส่คะแนน (0-100): คะแนน 85.50 -> เกรด B
ทำได้ดี พยายามต่อไป!
```

โปรแกรมนี้รวมเทคนิคที่เรียนมาแล้วทั้งหมด: การตรวจสอบ input ด้วย `scanf` และ
`clear_input_buffer` (จาก Part 3), Guard Clause สำหรับตรวจสอบช่วงคะแนน, `if-else if`
สำหรับแปลงคะแนนเป็นเกรด, และ `switch-case` สำหรับให้ข้อความตามเกรด — สังเกตว่าเราแยก
ตรรกะออกเป็นฟังก์ชันย่อย (`score_to_grade`, `is_valid_score`) แทนที่จะเขียนทุกอย่างรวม
กันใน `main` ซึ่งทำให้อ่านและทดสอบแต่ละส่วนแยกกันได้ง่ายขึ้นมาก (จะเจาะลึกเรื่องการออกแบบ
ฟังก์ชันใน Part 6)

## 4.8 ตัวอย่างโปรแกรมจริง: เมนูโปรแกรมด้วย switch (Step 32)

รูปแบบ **Menu-Driven Program** เป็นรูปแบบโปรแกรมที่พบบ่อยที่สุดรูปแบบหนึ่ง โดยเฉพาะ
โปรแกรมแบบ Command-Line ที่ให้ผู้ใช้เลือกจากรายการตัวเลือกซ้ำไปเรื่อยๆ (จะสมบูรณ์แบบเต็ม
รูปแบบเมื่อเรียนเรื่อง Loop ใน Part 5 ที่ทำให้เมนูวนกลับมาแสดงซ้ำได้จนกว่าผู้ใช้จะเลือกออก)

```c
/* simple_menu.c */
#include <stdio.h>

void clear_input_buffer(void) {
    int c;
    while ((c = getchar()) != '\n' && c != EOF) {
        /* discard */
    }
}

void show_menu(void) {
    printf("\n=== เมนูเครื่องคิดเลขอย่างง่าย ===\n");
    printf("1. บวก\n");
    printf("2. ลบ\n");
    printf("3. คูณ\n");
    printf("4. หาร\n");
    printf("เลือกเมนู (1-4): ");
}

int main(void) {
    int choice;
    double a, b;

    show_menu();

    if (scanf("%d", &choice) != 1) {
        printf("การเลือกเมนูไม่ถูกต้อง\n");
        return 1;
    }
    clear_input_buffer();

    switch (choice) {
        case 1:
        case 2:
        case 3:
        case 4:
            printf("ใส่จำนวนที่ 1: ");
            scanf("%lf", &a);
            printf("ใส่จำนวนที่ 2: ");
            scanf("%lf", &b);
            clear_input_buffer();
            break;
        default:
            printf("เมนูไม่ถูกต้อง กรุณาเลือก 1-4 เท่านั้น\n");
            return 1;
    }

    switch (choice) {
        case 1:
            printf("ผลลัพธ์: %.2f + %.2f = %.2f\n", a, b, a + b);
            break;
        case 2:
            printf("ผลลัพธ์: %.2f - %.2f = %.2f\n", a, b, a - b);
            break;
        case 3:
            printf("ผลลัพธ์: %.2f * %.2f = %.2f\n", a, b, a * b);
            break;
        case 4:
            if (b == 0.0) {
                printf("ข้อผิดพลาด: หารด้วยศูนย์ไม่ได้\n");
                return 1;
            }
            printf("ผลลัพธ์: %.2f / %.2f = %.2f\n", a, b, a / b);
            break;
        default:
            /* ไม่มีทางมาถึงจุดนี้ได้ เพราะกรองด้วย switch แรกไปแล้ว
               แต่ใส่ default ไว้เสมอตามหลักการที่เรียนในหัวข้อ 4.4 */
            break;
    }

    return 0;
}
```

```bash
gcc -Wall -Wextra -Wpedantic -std=c17 simple_menu.c -o simple_menu
printf "1\n15\n27\n" | ./simple_menu
```

ผลลัพธ์:

```

=== เมนูเครื่องคิดเลขอย่างง่าย ===
1. บวก
2. ลบ
3. คูณ
4. หาร
เลือกเมนู (1-4): ใส่จำนวนที่ 1: ใส่จำนวนที่ 2: ผลลัพธ์: 15.00 + 27.00 = 42.00
```

สังเกตว่าโปรแกรมนี้ตรวจสอบเงื่อนไข**การหารด้วยศูนย์**ไว้ด้วย (`if (b == 0.0)`) ก่อนที่จะ
คำนวณจริง นี่คือตัวอย่างของการรวม `if` เข้ากับ `switch` เพื่อจัดการ Edge Case ที่ `switch`
เพียงอย่างเดียวจัดการไม่ได้ (เพราะ `switch` เทียบได้แค่ค่าคงที่ ไม่สามารถเทียบกับเงื่อนไข
`b == 0.0` ซึ่งเป็นค่าที่รู้เฉพาะตอนรันโปรแกรมได้)

## ข้อผิดพลาดที่พบบ่อย (Common Pitfalls)

1. **ลืม `break` ใน switch-case** — ทำให้เกิด Fallthrough โดยไม่ตั้งใจ เป็นบั๊กคลาสสิกที่
   พบบ่อยที่สุดของ `switch` ดังตัวอย่างในหัวข้อ 4.3 ควรตรวจสอบทุก `case` ให้แน่ใจว่ามี
   `break` (หรือ `return`) ครบ ยกเว้นกรณีที่ตั้งใจให้ fallthrough จริงๆ

2. **เรียงลำดับ `else if` ผิด** — เขียนเงื่อนไขกว้างไว้ก่อนเงื่อนไขแคบ เช่น ตรวจ
   `score >= 60` ก่อน `score >= 90` ทำให้เงื่อนไขที่ตามมาไม่มีทางถูกตรวจสอบเลย (Dead Code)
   ต้องเรียงจากเงื่อนไขที่เฉพาะเจาะจงที่สุดไปหากว้างที่สุดเสมอ (หรือกลับกัน ขึ้นอยู่กับ
   ตรรกะของปัญหา แต่ต้องคิดให้รอบคอบ)

3. **ใช้ `switch` กับ `float`/`double`** — เป็น Compile Error ทันที เพราะมาตรฐานภาษา
   อนุญาตให้ `switch` ทำงานกับจำนวนเต็มเท่านั้น (การเปรียบเทียบ Floating Point ด้วย `==`
   ก็มีปัญหาเรื่องความแม่นยำอยู่แล้วในตัว ทำให้ `switch` กับทศนิยมไม่สมเหตุสมผลอยู่ดี)

4. **ซ้อน Ternary Operator หลายชั้นจนอ่านไม่ออก** — ดังตัวอย่างในหัวข้อ 4.5 ควรใช้ Ternary
   เฉพาะเงื่อนไขเดียวที่อ่านจบในบรรทัดเดียว ถ้าซับซ้อนกว่านั้นให้ใช้ `if-else if` แทน

5. **ลืมว่า `switch` ต้องการค่าคงที่ (Compile-time Constant) ใน case** — พยายามใช้ตัวแปร
   หรือช่วงค่า (เช่น `case x:` หรือ `case 1...10:` ซึ่งไม่ใช่มาตรฐาน C) ใน `case` จะทำให้
   คอมไพล์ไม่ผ่าน ถ้าต้องการเปรียบเทียบกับช่วงค่า ต้องใช้ `if-else if` เท่านั้น

6. **ปฏิเสธนิพจน์ตรรกะซ้อนกันจนอ่านยาก** — เขียน `!(a && b)` หรือ `!(a || !b)` โดยไม่แปลง
   ด้วย De Morgan's Law ให้อยู่ในรูปที่อ่านง่ายกว่า ทำให้ผู้อ่านโค้ด (รวมถึงตัวเองในอนาคต)
   ต้องนั่งแกะตรรกะปฏิเสธซ้อนปฏิเสธ เสี่ยงต่อการตีความผิดพลาด

## แบบฝึกหัดท้ายบท

1. เขียนโปรแกรมตรวจสอบว่าปีที่กำหนด (ตัวแปร `int` ในซอร์สโค้ด) เป็นปีอธิกสุรทิน
   (Leap Year) หรือไม่ ตามกฎ: หารด้วย 4 ลงตัว **และ** (หารด้วย 100 ไม่ลงตัว **หรือ**
   หารด้วย 400 ลงตัว) ทดสอบกับปี 2024, 2023, 1900, 2000

2. เขียนโปรแกรมที่ใช้ `switch` รับตัวเลข 1-12 แล้วพิมพ์ชื่อเดือนภาษาไทยที่ตรงกัน ถ้าไม่ใช่
   1-12 ให้พิมพ์ข้อความแจ้งว่าค่าไม่ถูกต้อง

3. เขียนโปรแกรมตรวจสอบเงื่อนไขการกู้เงิน: ผู้กู้ต้องมีอายุระหว่าง 20-60 ปี **และ** มีรายได้
   มากกว่า 15000 บาท จึงจะกู้ได้ ให้เขียนเงื่อนไขนี้สองแบบ (แบบตรงและแบบใช้ De Morgan's Law
   กับเงื่อนไขปฏิเสธ) แล้วเทียบผลลัพธ์ว่าตรงกันทุกกรณี

4. แปลงโค้ด Nested If ต่อไปนี้ให้เป็น Guard Clause ที่อ่านง่ายขึ้น (ต้องให้ผลลัพธ์เหมือนเดิม
   ทุกกรณี):

   ```c
   bool can_drive(int age, bool has_license, bool car_available) {
       if (age >= 18) {
           if (has_license) {
               if (car_available) {
                   return true;
               } else {
                   printf("ไม่มีรถ\n");
                   return false;
               }
           } else {
               printf("ไม่มีใบขับขี่\n");
               return false;
           }
       } else {
           printf("อายุไม่ถึงเกณฑ์\n");
           return false;
       }
   }
   ```

5. เขียนโปรแกรมเมนูอย่างง่าย (คล้ายหัวข้อ 4.8) ที่มี 3 ตัวเลือก: (1) แปลงเซลเซียสเป็น
   ฟาเรนไฮต์, (2) แปลงฟาเรนไฮต์เป็นเซลเซียส, (3) ออกจากโปรแกรม โดยใช้ `switch` เลือก
   การทำงาน (สูตร: `F = C * 9/5 + 32`, `C = (F - 32) * 5/9`)

6. เขียนฟังก์ชัน `char get_letter_grade(double score)` ที่ใช้ Ternary Operator **เพียง
   ครั้งเดียว** ผสมกับ `if-else if` (ไม่ใช่ซ้อน Ternary หลายชั้น) เพื่อคืนเกรดตัวอักษร
   พร้อมพิมพ์เครื่องหมาย `+`/`-` ต่อท้าย ถ้าเศษทศนิยมของคะแนนมากกว่า `.5` ให้ต่อท้ายด้วย
   `+` มิฉะนั้นไม่ต้องต่อท้าย (คำใบ้: ใช้ Ternary เฉพาะจุดตัดสิน `+`)

### แนวทางเฉลยข้อ 1

```c
/* leap_year_checker.c */
#include <stdio.h>
#include <stdbool.h>

bool is_leap_year(int year) {
    return (year % 4 == 0) && (year % 100 != 0 || year % 400 == 0);
}

int main(void) {
    int years[] = {2024, 2023, 1900, 2000};
    size_t count = sizeof(years) / sizeof(years[0]);

    for (size_t i = 0; i < count; i++) {
        int year = years[i];
        if (is_leap_year(year)) {
            printf("ปี %d เป็นปีอธิกสุรทิน\n", year);
        } else {
            printf("ปี %d ไม่ใช่ปีอธิกสุรทิน\n", year);
        }
    }

    return 0;
}
```

```bash
gcc -Wall -Wextra -Wpedantic -std=c17 leap_year_checker.c -o leap_year_checker
./leap_year_checker
```

ผลลัพธ์:

```
ปี 2024 เป็นปีอธิกสุรทิน
ปี 2023 ไม่ใช่ปีอธิกสุรทิน
ปี 1900 ไม่ใช่ปีอธิกสุรทิน
ปี 2000 เป็นปีอธิกสุรทิน
```

จุดที่น่าสนใจคือปี 1900 หารด้วย 4 ลงตัว แต่**ไม่ใช่**ปีอธิกสุรทิน เพราะหารด้วย 100 ลงตัว
และหารด้วย 400 **ไม่ลงตัว** (`1900 / 400 = 4.75`) ในขณะที่ปี 2000 หารด้วย 400 ลงตัวพอดี
จึงเป็นปีอธิกสุรทิน — นี่คือตัวอย่างจริงที่แสดงให้เห็นว่าทำไมกฎปีอธิกสุรทินต้องมีเงื่อนไข
ซ้อนกันถึง 3 ชั้น (ไม่ใช่แค่ "หารด้วย 4 ลงตัว" เฉยๆ) และทำไมการรวมตัวดำเนินการตรรกะ
`&&`/`||` ให้ถูกต้องตาม Operator Precedence จึงสำคัญมาก (`&&` มีลำดับสูงกว่า `||` ทำให้
นิพจน์นี้ทำงานถูกต้องแม้ไม่ใส่วงเล็บรอบ `year % 100 != 0 || year % 400 == 0` แต่การใส่
วงเล็บไว้ก็ยังช่วยให้อ่านง่ายขึ้นตามหลักที่เรียนใน Part 2)

### แนวทางเฉลยข้อ 4

```c
/* guard_clause_solution.c */
#include <stdio.h>
#include <stdbool.h>

bool can_drive(int age, bool has_license, bool car_available) {
    if (age < 18) {
        printf("อายุไม่ถึงเกณฑ์\n");
        return false;
    }
    if (!has_license) {
        printf("ไม่มีใบขับขี่\n");
        return false;
    }
    if (!car_available) {
        printf("ไม่มีรถ\n");
        return false;
    }

    return true;
}

int main(void) {
    printf("%s\n", can_drive(20, true, true)  ? "ขับได้"   : "ขับไม่ได้");
    printf("%s\n", can_drive(16, true, true)  ? "ขับได้"   : "ขับไม่ได้");
    printf("%s\n", can_drive(20, false, true) ? "ขับได้"   : "ขับไม่ได้");
    printf("%s\n", can_drive(20, true, false) ? "ขับได้"   : "ขับไม่ได้");

    return 0;
}
```

```bash
gcc -Wall -Wextra -Wpedantic -std=c17 guard_clause_solution.c -o guard_clause_solution
./guard_clause_solution
```

ผลลัพธ์:

```
ขับได้
อายุไม่ถึงเกณฑ์
ขับไม่ได้
ไม่มีใบขับขี่
ขับไม่ได้
ไม่มีรถ
ขับไม่ได้
```

สังเกตว่าการแปลง Nested If เป็น Guard Clause ทำได้โดยการ **กลับเงื่อนไขแต่ละชั้นให้เป็น
เงื่อนไข "ล้มเหลว" แล้ว `return` ออกทันที** (`age >= 18` กลายเป็น `age < 18` แล้วออกก่อน,
`has_license` กลายเป็น `!has_license` แล้วออกก่อน) ผลลัพธ์คือฟังก์ชันที่มีความลึกของ
วงเล็บปีกกาแค่ชั้นเดียวตลอดทั้งฟังก์ชัน ไม่ว่าจะมีเงื่อนไขกี่ข้อก็ตาม อ่านและแก้ไขเพิ่มเติม
ในอนาคตได้ง่ายกว่ามาก

## สรุปท้ายบท

ใน Part นี้เราได้:

- เข้าใจกลไก `if/else if/else` และแนวคิดเรื่อง "จริง/เท็จ" ของ C ที่ตัดสินจากค่าตัวเลข
  (ไม่ใช่ศูนย์ = จริงเสมอ) พร้อมรู้จัก `<stdbool.h>` ที่ทำให้โค้ดสื่อเจตนาชัดเจนขึ้น
- เรียนรู้เทคนิค Guard Clause ที่ทำให้เงื่อนไขซ้อนกันหลายชั้นอ่านง่ายขึ้นอย่างมาก
- ใช้ `switch-case` ได้อย่างถูกต้อง เข้าใจกลไก Fallthrough ทั้งที่เป็นบั๊กและที่ตั้งใจ
  พร้อมรู้ข้อจำกัดว่า `switch` ใช้ได้กับจำนวนเต็มและค่าคงที่เท่านั้น
- ใช้ Ternary Operator แทนเงื่อนไขสั้นๆ ได้อย่างเหมาะสม โดยไม่ซ้อนจนอ่านไม่ออก
- ใช้ De Morgan's Law แปลงเงื่อนไขปฏิเสธซ้อนกันให้อยู่ในรูปที่อ่านง่ายขึ้น
- สร้างโปรแกรมจริงสองตัวอย่าง (เครื่องคิดเกรด, เมนูโปรแกรม) ที่รวมทุกเทคนิคเข้าด้วยกัน

โปรแกรมทำงานซ้ำๆ ได้โดยอัตโนมัติ เป็นอิฐก้อนสำคัญที่จะนำไปสู่การเขียนโปรแกรมที่ประมวลผล
